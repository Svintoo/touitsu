cmake_minimum_required(VERSION 3.20)
project(TouitsuHook LANGUAGES C CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(TX "${CMAKE_CURRENT_LIST_DIR}/../../third_party/Textractor")
include_directories("${TX}/include")
add_compile_definitions(UNICODE _UNICODE ENGLISH VERSION="touitsu")
add_library(text STATIC "${TX}/text.cpp")
target_precompile_headers(text PRIVATE "${TX}/include/common.h")
add_subdirectory("${TX}/texthook" texthook)
# Suspended game threads can still return through original-code trampolines.
# Keep those executable slots alive after removing a hook entry.
file(READ "${TX}/texthook/minhook/src/hook.c" minhook_code)
string(FIND "${minhook_code}" "FreeBuffer(g_hooks.pItems[pos].pTrampoline);" retire_patch_position)
if(retire_patch_position EQUAL -1)
  message(FATAL_ERROR "MinHook changed: review the trampoline lifetime patch before updating")
endif()
string(REPLACE "FreeBuffer(g_hooks.pItems[pos].pTrampoline);" "/* Retired until process exit; a game thread may still use this code. */" minhook_code "${minhook_code}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/hook_retired.c" "${minhook_code}")
get_target_property(minhook_sources minhook SOURCES)
list(REMOVE_ITEM minhook_sources minhook/src/hook.c)
set_property(TARGET minhook PROPERTY SOURCES "${minhook_sources}")
target_sources(minhook PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/hook_retired.c")
target_include_directories(minhook PRIVATE "${TX}/texthook/minhook/src")
get_target_property(hook_sources texthook SOURCES)
list(REMOVE_ITEM hook_sources hookfinder.cc)
set_property(TARGET texthook PROPERTY SOURCES "${hook_sources}")
target_sources(texthook PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../tx_hookfinder.cpp")
target_compile_options(texthook PRIVATE /std:c++17 /wd4018 "/FI${CMAKE_CURRENT_LIST_DIR}/../tx_hook_prefix.h")
target_compile_options(text PRIVATE /std:c++17)
target_link_libraries(texthook text)
set_target_properties(texthook PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${TOUITSU_OUTPUT}" LIBRARY_OUTPUT_DIRECTORY_RELEASE "${TOUITSU_OUTPUT}")
